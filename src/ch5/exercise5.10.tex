\item
\[
    \Re F(x)
    = \Re F(x_0)
    + \frac{x - x_0}{\pi} \Pint_{-\infty}^\infty dx' \frac{\Im F(x')}{(x' - x_0)(x' - x)}
\]
\begin{align*}
    \Rightarrow \Re F(0)
    &= \Re F(x_0) - \inv{\pi} \Pint_{-\infty}^\infty dx' \Im F(x') \left(
        \frac{x_0}{x' (x' - x_0)}
    \right) \\
    &\xrightarrow{x_0 \rightarrow \pm \infty}
        \inv{\pi} \Pint_{-\infty}^\infty dx' \frac{\Im F(x')}{x'}
\end{align*}

\item
We start with the unsubtracted dispersion relation.
\begin{align*}
    \Re xF(x)
    &= \inv{\pi} \Pint_{-\infty}^\infty dx' \frac{\Im F(x')}{\frac{x'}{x} - 1} \\
    &= -\inv{\pi} \Pint_{-\infty}^\infty dx' \Im F(x') \left( 1 + \frac{x'}{x} + \frac{x'^2}{x^2} \cdots \right).
\end{align*}
Therefore, in the limit $x \rightarrow \pm \infty$, one obtains the superconvergence relation
\[
    \Pint_{-\infty}^\infty dx' \Im F(x') = 0.
\]

\end{enumerate}
